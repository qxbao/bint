package hookcore
import("testing"; "bint/internal/config")
func TestAcceptanceInvestigationReadonly(t *testing.T) {
 cfg,_,err:=config.Load("<repo>",true);if err!=nil {t.Fatal(err)}
 cases:=[]struct{name,cmd string}{
 {"literal-pipe",`grep -n "mcpCheckTimeout\|context.With" internal/app/mcp_svc.go`},
 {"ls-files",`git ls-files internal/mcpcat/probe`},
 {"plain-grep",`grep -n "NT-MCP-PROBE" docs/acceptance.md`},
 {"log-regex",`grep -n -E "^(--- FAIL|FAIL|ok|panic)|CopyCodegraph" /tmp/task/verify-1.log`},
 {"log-simple",`grep -n FAIL /tmp/task/verify-1.log`},
 {"actual-pipe",`git log | sh`},
 }
 for _,c:=range cases {t.Run(c.name,func(t *testing.T){
  a,why:=readonlyShellAllowed(c.cmd,config.Default().Workflow.Agent.ReadonlyTools);b,why2:=readonlyShellAllowed(c.cmd,cfg.Workflow.Agent.ReadonlyTools)
  t.Logf("default allow=%v reason=%q; project allow=%v reason=%q; command=%s",a,why,b,why2,c.cmd)
  if b {t.Errorf("historical reproduction expected rejection by current project rules")}
  if c.name=="plain-grep"||c.name=="log-simple" {if !a {t.Errorf("default grep should be allowed")}} else if a {t.Errorf("expected rejection under defaults")}
 })}
}
